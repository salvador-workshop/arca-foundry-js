    math,
    position,

const interlockDovetailPanelModule = require('./interlock-dovetail-panel/index.js')
const interlockDovetailPanel = interlockDovetailPanelModule.init({ jscad, swcadJs })

const interlockDovetailFrameModule = require('./interlock-dovetail-frame/index.js')
const interlockDovetailFrame = interlockDovetailFrameModule.init({ jscad, swcadJs })

const groovedShelfModule = require('./grooved-shelf/index.js')
const groovedShelf = groovedShelfModule.init({ jscad, swcadJs })

const groovedBackingModule = require('./grooved-backing/index.js')

const spaceUnit = math.inchesToMm(4.5)

/**
 * Main function to be rendered by JSCAD
 * @returns `geom3` or `geom3[]`
 */
function main() {
    const interDovetailFrameOpts = {
        size: [math.inchesToMm(2.5), math.inchesToMm(3.5)],
        panelThickness: 1.3333,
        postThickness: 4,
        dovetailLength: math.inchesToMm(1 / 2),
    }

    const interDovetailFrame1 = interlockDovetailFrame({
        ...interDovetailFrameOpts,
        beamLevels: [15, math.inchesToMm(3.5)],
        positionType: 'face',
    })[0]

    const interDovetailFrame2 = interlockDovetailFrame({
        ...interDovetailFrameOpts,
        positionType: 'side',
    })[0]

    const interDovetailFrame3 = interlockDovetailFrame({
        ...interDovetailFrameOpts,
        positionType: 'face',
    })[0]

    const interDovetailFrame4 = interlockDovetailFrame({
        ...interDovetailFrameOpts,
        positionType: 'side',
    })[0]

    const interlockDovetailPanelOpts = {
        size: [math.inchesToMm(2.5), math.inchesToMm(3.5)],
        panelThickness: 1.3333,
        postThickness: 4,
        dovetailLength: math.inchesToMm(1 / 2),
    }

    const interlockDovetailPanel1 = interlockDovetailPanel({
        ...interlockDovetailPanelOpts,
        bodyType: 'solid',
        positionType: 'face',
    })[0]
    const interlockDovetailPanel2 = interlockDovetailPanel({
        ...interlockDovetailPanelOpts,
        bodyType: 'mesh',
        positionType: 'side',
        panelLevels: [
            [20, 50]
        ]
    })[0]

    const interlockDovetailPanel3 = interlockDovetailPanel({
        ...interlockDovetailPanelOpts,
        bodyType: 'minimal',
        positionType: 'face',
        panelLevels: [
            [11, math.inchesToMm(3.5)]
        ]
    })[0]

    const interlockDovetailPanel4 = interlockDovetailPanel({
        ...interlockDovetailPanelOpts,
        bodyType: 'minimal',
        positionType: 'side',
        panelLevels: [
            [math.inchesToMm(0.5), math.inchesToMm(2)],
            [math.inchesToMm(2.25), math.inchesToMm(3.5)],
        ]
    })[0]

    const interlockDovetailPanel5 = interlockDovetailPanel({
        ...interlockDovetailPanelOpts,
        bodyType: 'minimal',
        positionType: 'side',
        panelLevels: [
            [math.inchesToMm(0.5), math.inchesToMm(2)],
            [math.inchesToMm(2.25), math.inchesToMm(3.5)],
        ],
        omitBottomPanelBeam: true,
    })[0]

    const groovedShelfOpts = {
        supportSize: [math.inchesToMm(2.5), math.inchesToMm(2.5) + 4],
        thickness: 1.3333,
        postThickness: 4,
        panelType: 'minimal',
    }

    const gShelf1 = groovedShelf({
        ...groovedShelfOpts,
        type: 'wrap'
    })[0]

    const gShelf2 = groovedShelf({
        ...groovedShelfOpts,
        type: 'flush'
    })[0]

    const gShelf3 = groovedShelf({
        ...groovedShelfOpts,
        panelType: 'lattice',
        // type: 'flush',
        isOpen: true,
    })[0]

    const gShelf4 = groovedShelf({
        ...groovedShelfOpts,
        panelType: 'lattice',
        type: 'flush',
        hasCutCorners: true,
        cornerSize: [4, 6]
    })[0]

    const gShelf5 = groovedShelf({
        ...groovedShelfOpts,
        panelType: 'minimal',
        type: 'wrap',
        hasCutCorners: true,
        cornerSize: [4, 6]
    })[0]

    const groovedBackingOpts = {
        supportSpan: math.inchesToMm(2.5),
        backingWidth: math.inchesToMm(3.5),
        thickness: 1.3333,
        postThickness: 4,
        panelType: 'minimal',
    }

    const gBacking1 = groovedBacking({
        ...groovedBackingOpts,
        type: 'wrap'
    })[0]

    const gBacking2 = groovedBacking({
        ...groovedBackingOpts,
        type: 'flush'
    })[0]

    const gBacking3 = groovedBacking({
        ...groovedBackingOpts,
        type: 'lip'
    })[0]

    const gBacking4 = groovedBacking({
        ...groovedBackingOpts,
        type: 'flush',
        hasBottomLip: true,
    })[0]

    const gBacking5 = groovedBacking({
        ...groovedBackingOpts,
        type: 'lip',
        hasBottomLip: true,
    })[0]

    const modelRow1 = [
        // dPost1,
        // dPost2,
        interDovetailFrame1,
        interDovetailFrame2,
        interDovetailFrame3,
        interDovetailFrame4,
    ]

    const modelRow2 = [
        interlockDovetailPanel1,
        interlockDovetailPanel2,
        interlockDovetailPanel3,
        interlockDovetailPanel4,
        interlockDovetailPanel5,
    ]

    const modelRow3 = [
        gShelf1,
        gShelf2,
        gShelf3,
        gShelf4,
        gShelf5,
    ]

    const modelRow4 = [
        gBacking1,
        gBacking2,
        gBacking3,
        gBacking4,
        gBacking5,
    ]

    const outModels1 = modelRow1.map((oModel, idx) => {
        return translate([idx * spaceUnit, 0, 0], oModel)
    })

    const outModels2 = modelRow2.map((oModel, idx) => {
        return translate([idx * spaceUnit, spaceUnit, 0], oModel)
    })

    const outModels3 = modelRow3.map((oModel, idx) => {
        return translate([idx * spaceUnit, spaceUnit * 2, 0], oModel)
    })

    const outModels4 = modelRow4.map((oModel, idx) => {
        return translate([idx * spaceUnit, spaceUnit * 3, 0], oModel)
    })

    return [
        ...outModels1,
        ...outModels2,
        ...outModels3,
        ...outModels4,
    ]

    // return gShelf2
}


module.exports = {
    main,
}
